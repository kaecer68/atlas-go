package narrative

import (
    "fmt"
    "strings"
    
    "github.com/kaecer68/atlas-go/internal/domain"
)

// ReportGenerator composes natural language reports from system data
type ReportGenerator struct {
    templates NLGTemplates
}

// NLGTemplates holds all template strings
type NLGTemplates struct {
    MarketOverview     string
    NarrativeSummary   string
    TopPickIntro       string
    FactorExplanation  string
    RiskWarning        string
    ControlDecision    string
}

func NewReportGenerator() *ReportGenerator {
    return &ReportGenerator{
        templates: NLGTemplates{
            MarketOverview:    TemplateMarketOverview,
            NarrativeSummary:  TemplateNarrativeSummary,
            TopPickIntro:      TemplateTopPickIntro,
            FactorExplanation: TemplateFactorExplanation,
            RiskWarning:       TemplateRiskWarning,
            ControlDecision:   TemplateControlDecision,
        },
    }
}

func (rg *ReportGenerator) GenerateDailySummary(
    date string,
    events []domain.NarrativeEvent,
    recommendations []domain.Recommendation,
    risk domain.RiskSnapshot,
) (*domain.DailySummaryReport, error) {
    sections := []domain.ReportSection{}
    
    // Section 1: Market Overview
    trend := "平盤震盪"
    if len(events) > 0 {
        trend = "受宏觀事件影響"
    }
    sections = append(sections, domain.ReportSection{
        Title:   "市場概覽",
        Content: fmt.Sprintf("今日大盤%s。", trend),
        Priority: 1,
    })
    
    // Section 2: Narrative Events
    if len(events) > 0 {
        var eventNames []string
        for _, e := range events {
            eventNames = append(eventNames, e.Name)
        }
        sections = append(sections, domain.ReportSection{
            Title:   "宏觀敘事",
            Content: fmt.Sprintf("偵測到%d個宏觀敘事事件：%s。", len(events), strings.Join(eventNames, "、")),
            Priority: 2,
        })
    }
    
    // Section 3: Top Recommendations
    if len(recommendations) > 0 {
        sections = append(sections, domain.ReportSection{
            Title:   "推薦標的",
            Content: fmt.Sprintf("系統推薦關注%d檔標的。", len(recommendations)),
            Priority: 3,
        })
    }
    
    // Section 4: Risk Assessment
    riskLevel := "低"
    if risk.VaR95 > 0.05 {
        riskLevel = "高"
    } else if risk.VaR95 > 0.03 {
        riskLevel = "中"
    }
    sections = append(sections, domain.ReportSection{
        Title:   "風險評估",
        Content: fmt.Sprintf("當前投組VaR(95%%)為%.2f%%，風險等級：%s。", risk.VaR95*100, riskLevel),
        Priority: 4,
    })
    
    return &domain.DailySummaryReport{
        Date:       date,
        Sections:   sections,
        TopPicks:   recommendations,
        RiskLevel:  riskLevel,
        NarrativeCount: len(events),
    }, nil
}

    rec domain.Recommendation,
) string {
    if rec.FactorScores == nil {
        return fmt.Sprintf("%s：無因子分數資料。", rec.Symbol)

        var rationale string
        switch name {
        case "momentum":
            rationale = "近期動能表現"
        case "value":
            rationale = "估值吸引力"
        case "quality":
            rationale = "財務品質"
        default:
            rationale = "綜合評估"
        }
        parts = append(parts, fmt.Sprintf("%s因子 %.1f 分（%s）", name, score, rationale))
    }
    
    return strings.Join(parts, "；")
}

func (api *DashboardAPI) handleDailySummary(w http.ResponseWriter, r *http.Request) {
    ctx := r.Context()
    
    // Fetch latest narrative events
    events, err := api.narrativeEngine.DetectEvents(ctx, api.macroProvider.GetSnapshot())
    if err != nil {
        http.Error(w, fmt.Sprintf("failed to detect events: %v", err), http.StatusInternalServerError)
        return
    }
    
    // Fetch latest recommendations
    recs, err := api.getLatestRecommendations(ctx)
    if err != nil {
        http.Error(w, fmt.Sprintf("failed to get recommendations: %v", err), http.StatusInternalServerError)
        return
    }
    
    // Get current risk snapshot
    risk := api.getCurrentRiskSnapshot()
    
    // Generate report
    report, err := api.reportGenerator.GenerateDailySummary(
        time.Now().Format("2006-01-02"),
        events,
        recs,
        risk,
    )
    if err != nil {
        http.Error(w, fmt.Sprintf("failed to generate report: %v", err), http.StatusInternalServerError)
        return
    }
    
    w.Header().Set("Content-Type", "application/json; charset=utf-8")
    json.NewEncoder(w).Encode(report)
}

package tax

import (
    "github.com/kaecer68/atlas-go/internal/domain"
)

// TaiwanTaxCalculator computes Taiwan-specific taxes
type TaiwanTaxCalculator struct {
    config domain.TaxConfig
}

func NewTaiwanTaxCalculator(config domain.TaxConfig) *TaiwanTaxCalculator {
    return &TaiwanTaxCalculator{config: config}
}

// CalculateDividendTax returns tax on dividend income
func (c *TaiwanTaxCalculator) CalculateDividendTax(dividendAmount float64) float64 {
    return dividendAmount * c.config.DividendTaxRate
}

// CalculateTransactionTax returns securities transaction tax
// In Taiwan, this applies to sell-side only
func (c *TaiwanTaxCalculator) CalculateTransactionTax(sellNotional float64) float64 {
    return sellNotional * c.config.TransactionTaxRate
}

// CalculatePositionTax computes total tax for a position
func (c *TaiwanTaxCalculator) CalculatePositionTax(
    pos domain.Position,
    sellPrice float64,
    dividendReceived float64,
) domain.TaxSnapshot {
    sellNotional := sellPrice * float64(pos.Quantity)
    
    dividendTax := c.CalculateDividendTax(dividendReceived)
    transactionTax := c.CalculateTransactionTax(sellNotional)
    totalTax := dividendTax + transactionTax
    
    grossPnL := sellNotional - (pos.AvgPrice * float64(pos.Quantity))
    afterTaxPnL := grossPnL - totalTax
    
    return domain.TaxSnapshot{
        Symbol:           pos.Symbol,
        DividendTaxRate:  c.config.DividendTaxRate,
        TransactionTaxRate: c.config.TransactionTaxRate,
        DividendTax:      dividendTax,
        TransactionTax:   transactionTax,
        TotalTax:         totalTax,
        AfterTaxPnL:      afterTaxPnL,
    }
}

// CalculatePortfolioTax computes aggregate tax for all positions
func (c *TaiwanTaxCalculator) CalculatePortfolioTax(
    positions []domain.Position,
    sellPrices map[string]float64,
    dividends map[string]float64,
) []domain.TaxSnapshot {
    snapshots := make([]domain.TaxSnapshot, 0, len(positions))
    
    for _, pos := range positions {
        sellPrice := sellPrices[pos.Symbol]
        if sellPrice == 0 {
            sellPrice = pos.AvgPrice // default to no gain/loss
        }
        dividend := dividends[pos.Symbol]
        
        snapshot := c.CalculatePositionTax(pos, sellPrice, dividend)
        snapshots = append(snapshots, snapshot)
    }
    
    return snapshots
}
```

package tax

import (
    "github.com/kaecer68/atlas-go/internal/domain"
    "github.com/kaecer68/atlas-go/internal/portfolio"
)

// TaxAwareSizer wraps a base sizer and adjusts for transaction tax
type TaxAwareSizer struct {
    baseSizer portfolio.Sizer
    taxConfig domain.TaxConfig
}

func NewTaxAwareSizer(baseSizer portfolio.Sizer, config domain.TaxConfig) *TaxAwareSizer {
    return &TaxAwareSizer{
        baseSizer: baseSizer,
        taxConfig: config,
    }
}

// SizePosition reduces capital by expected transaction tax before sizing
func (s *TaxAwareSizer) SizePosition(symbol string, capital float64, price float64) int {
    // Reserve capital for transaction tax (sell-side)
    // Assuming round-trip: buy now, sell later at same price
    // Tax applies to sell notional = shares * price
    // sellNotional = capital (approximately)
    // tax = sellNotional * rate
    // effectiveCapital = capital - tax
    // But tax depends on sellNotional which depends on shares...
    // Approximate: reduce capital by tax rate
    effectiveCapital := capital * (1 - s.taxConfig.TransactionTaxRate)
    return s.baseSizer.SizePosition(symbol, effectiveCapital, price)
}
```

package monitoring

import (
    "bufio"
    "encoding/json"
    "fmt"
    "os"
    "sync"
    "time"

// AlertStore persists alerts to JSONL file
type AlertStore struct {
    path string
    mu   sync.RWMutex
}

func NewAlertStore(path string) *AlertStore {
    return &AlertStore{path: path}
}

func (s *AlertStore) Save(alert domain.AlertRecord) error {
    s.mu.Lock()
    defer s.mu.Unlock()
    
    f, err := os.OpenFile(s.path, os.O_APPEND|os.O_CREATE|os.O_WRONLY, 0644)
    if err != nil {
        return fmt.Errorf("open alert store: %w", err)
    }
    defer f.Close()
    
    line, err := json.Marshal(alert)
    if err != nil {
        return fmt.Errorf("marshal alert: %w", err)
    }
    
    if _, err := f.WriteString(string(line) + "\n"); err != nil {
        return fmt.Errorf("write alert: %w", err)
    }
    
    return nil
}

func (s *AlertStore) LoadAll() ([]domain.AlertRecord, error) {
    s.mu.RLock()
    defer s.mu.RUnlock()
    
    f, err := os.Open(s.path)
    if err != nil {
        if os.IsNotExist(err) {
            return []domain.AlertRecord{}, nil
        }
        return nil, fmt.Errorf("open alert store: %w", err)
    }
    defer f.Close()
    
    var alerts []domain.AlertRecord
    scanner := bufio.NewScanner(f)
    for scanner.Scan() {
        var alert domain.AlertRecord
        if err := json.Unmarshal(scanner.Bytes(), &alert); err != nil {
            continue // skip malformed lines
        }
        alerts = append(alerts, alert)
    }
    
    return alerts, scanner.Err()
}

func (s *AlertStore) LoadUnacknowledged() ([]domain.AlertRecord, error) {
    alerts, err := s.LoadAll()
    if err != nil {
        return nil, err
    }
    
    var unack []domain.AlertRecord
    for _, a := range alerts {
        if !a.Acknowledged {
            unack = append(unack, a)
        }
    }
    return unack, nil
}

func (s *AlertStore) Acknowledge(alertID string, user string) error {
    // In append-only store, we append an acknowledgment record
    // Or use in-memory index + rewrite (simpler for now)
    alerts, err := s.LoadAll()
    if err != nil {
        return err
    }
    
    now := time.Now()
    for i := range alerts {
        if alerts[i].ID == alertID {
            alerts[i].Acknowledged = true
            alerts[i].AcknowledgedAt = &now
            alerts[i].AcknowledgedBy = user
        }
    }
    
    // Rewrite file
    s.mu.Lock()
    defer s.mu.Unlock()
    
    f, err := os.Create(s.path)
    if err != nil {
        return fmt.Errorf("rewrite alert store: %w", err)
    }
    defer f.Close()
    
    for _, alert := range alerts {
        line, _ := json.Marshal(alert)
        f.WriteString(string(line) + "\n")
    }
    
    return nil
}

    if err != nil {

        return
    }
    
    w.Header().Set("Content-Type", "application/json")
    json.NewEncoder(w).Encode(alerts)
}

    if err != nil {
        http.Error(w, err.Error(), http.StatusInternalServerError)
        return
    }
    
    w.Header().Set("Content-Type", "application/json")
    json.NewEncoder(w).Encode(alerts)
}

func (api *AlertAPI) handleAcknowledgeAlert(w http.ResponseWriter, r *http.Request) {
    if r.Method != http.MethodPost {
        http.Error(w, "method not allowed", http.StatusMethodNotAllowed)
        return
    }
    
    var req struct {
        AlertID string `json:"alert_id"`
        User    string `json:"user"`
    }
    
    if err := json.NewDecoder(r.Body).Decode(&req); err != nil {
        http.Error(w, err.Error(), http.StatusBadRequest)
        return
    }
    
    if err := api.store.Acknowledge(req.AlertID, req.User); err != nil {
        http.Error(w, err.Error(), http.StatusInternalServerError)
        return
    }
    
    w.WriteHeader(http.StatusOK)
    json.NewEncoder(w).Encode(map[string]string{"status": "acknowledged"})
}

import (

// CapitalPhaseController manages capital deployment phases
type CapitalPhaseController struct {
    config   domain.CapitalPhaseConfig
    snapshot domain.CapitalSnapshot
}

func NewCapitalPhaseController(config domain.CapitalPhaseConfig) *CapitalPhaseController {
    return &CapitalPhaseController{
        config: config,
        snapshot: domain.CapitalSnapshot{
            Phase:          config.CurrentPhase,
            PhaseStartDate: config.PhaseStartDate,
            DaysInPhase:    int(time.Since(config.PhaseStartDate).Hours() / 24),
        },
    }
}

func (c *CapitalPhaseController) GetSnapshot() domain.CapitalSnapshot {
    c.snapshot.DaysInPhase = int(time.Since(c.config.PhaseStartDate).Hours() / 24)
    return c.snapshot
}

func (c *CapitalPhaseController) UpdateMetrics(rollingSharpe, maxDrawdown float64) {
    c.snapshot.RollingSharpe = rollingSharpe
    c.snapshot.MaxDrawdown = maxDrawdown
}

func (c *CapitalPhaseController) CanAdvance() (bool, string) {
    daysInPhase := int(time.Since(c.config.PhaseStartDate).Hours() / 24)
    
    if daysInPhase < c.config.MinDaysPerPhase {
        return false, fmt.Sprintf(
            "insufficient days in phase: %d < %d minimum",
            daysInPhase,
            c.config.MinDaysPerPhase,
        )
    }
    
    if c.snapshot.MaxDrawdown > c.config.MaxDrawdownLimit {
        return false, fmt.Sprintf(
            "max drawdown %.2f%% exceeds limit %.2f%%",
            c.snapshot.MaxDrawdown*100,
            c.config.MaxDrawdownLimit*100,
        )
    }
    
    if c.snapshot.RollingSharpe < c.config.SharpeThreshold {
        return false, fmt.Sprintf(
            "rolling Sharpe %.2f below threshold %.2f",
            c.snapshot.RollingSharpe,
            c.config.SharpeThreshold,
        )
    }
    
    return true, "all criteria met"
}

func (c *CapitalPhaseController) AdvancePhase() error {
    canAdvance, reason := c.CanAdvance()
    if !canAdvance {
        return fmt.Errorf("cannot advance phase: %s", reason)
    }
    
    var nextPhase domain.CapitalPhase
    switch c.config.CurrentPhase {
    case domain.PhaseSimulation:
        nextPhase = domain.PhasePaper
    case domain.PhasePaper:
        nextPhase = domain.PhaseLive
    case domain.PhaseLive:
        nextPhase = domain.PhaseFull
    case domain.PhaseFull:
        return fmt.Errorf("already at maximum phase")
    }
    
    c.config.CurrentPhase = nextPhase
    c.config.PhaseStartDate = time.Now()
    c.snapshot.Phase = nextPhase
    c.snapshot.PhaseStartDate = time.Now()
    c.snapshot.DaysInPhase = 0
    
    return nil
}

func (c *CapitalPhaseController) GetCapitalLimit() float64 {
    limit, ok := c.config.CapitalLimits[string(c.config.CurrentPhase)]
    if !ok {
        return 1.0 // default to 100%
    }
    return limit
}

func (c *CapitalPhaseController) CalculateMaxPositionSize(totalCapital float64) float64 {
    limit := c.GetCapitalLimit()
    return totalCapital * limit
}

// ApprovalRequest represents a human approval request
type ApprovalRequest struct {
    ID          string    `json:"id"`
    Type        string    `json:"type"`        // e.g., "live-to-full"
    Status      string    `json:"status"`      // pending/approved/rejected
    RequestedBy string    `json:"requested_by"`
    RequestedAt time.Time `json:"requested_at"`
    Reason      string    `json:"reason"`
    ApprovedBy  string    `json:"approved_by,omitempty"`
    ApprovedAt  *time.Time `json:"approved_at,omitempty"`
    ApprovalNote string    `json:"approval_note,omitempty"`
}

// ApprovalWorkflow manages human-in-the-loop approvals
type ApprovalWorkflow struct {
    path string
    mu   sync.RWMutex
}

func NewApprovalWorkflow(path string) *ApprovalWorkflow {
    return &ApprovalWorkflow{path: path}
}

func (w *ApprovalWorkflow) RequestApproval(reqType, requestedBy, reason string) (*ApprovalRequest, error) {
    req := &ApprovalRequest{
        ID:          uuid.New().String(),
        Type:        reqType,
        Status:      "pending",
        RequestedBy: requestedBy,
        RequestedAt: time.Now(),
        Reason:      reason,
    }
    
    w.mu.Lock()
    defer w.mu.Unlock()
    
    f, err := os.OpenFile(w.path, os.O_APPEND|os.O_CREATE|os.O_WRONLY, 0644)
    if err != nil {
        return nil, fmt.Errorf("open approval store: %w", err)
    }
    defer f.Close()
    
    line, _ := json.Marshal(req)
    if _, err := f.WriteString(string(line) + "\n"); err != nil {
        return nil, fmt.Errorf("write approval request: %w", err)
    }
    
    return req, nil
}

func (w *ApprovalWorkflow) Approve(requestID, approvedBy, note string) error {
    requests, err := w.LoadAll()
    if err != nil {
        return err
    }
    
    now := time.Now()
    found := false
    for i := range requests {
        if requests[i].ID == requestID {
            requests[i].Status = "approved"
            requests[i].ApprovedBy = approvedBy
            requests[i].ApprovedAt = &now
            requests[i].ApprovalNote = note
            found = true
            break
        }
    }
    
    if !found {
        return fmt.Errorf("approval request %s not found", requestID)
    }
    
    // Rewrite file
    w.mu.Lock()
    defer w.mu.Unlock()
    
    f, err := os.Create(w.path)
    if err != nil {
        return fmt.Errorf("rewrite approval store: %w", err)
    }
    defer f.Close()
    
    for _, req := range requests {
        line, _ := json.Marshal(req)
        f.WriteString(string(line) + "\n")
    }
    
    return nil
}

func (w *ApprovalWorkflow) LoadAll() ([]ApprovalRequest, error) {
    w.mu.RLock()
    defer w.mu.RUnlock()
    
    f, err := os.Open(w.path)
    if err != nil {
        if os.IsNotExist(err) {
            return []ApprovalRequest{}, nil
        }
        return nil, fmt.Errorf("open approval store: %w", err)
    }
    defer f.Close()
    
    var requests []ApprovalRequest
    scanner := bufio.NewScanner(f)
    for scanner.Scan() {
        var req ApprovalRequest
        if err := json.Unmarshal(scanner.Bytes(), &req); err != nil {
            continue
        }
        requests = append(requests, req)
    }
    
    return requests, scanner.Err()
}

func (w *ApprovalWorkflow) GetRequest(requestID string) (*ApprovalRequest, error) {
    requests, err := w.LoadAll()
    if err != nil {
        return nil, err
    }
    
    for i := range requests {
        if requests[i].ID == requestID {
            return &requests[i], nil
        }
    }
    
    return nil, fmt.Errorf("approval request %s not found", requestID)
}

package portfolio

import (
    "github.com/kaecer68/atlas-go/internal/domain"
)

// CapitalAllocation represents the result of capital allocation
type CapitalAllocation struct {
    TotalCapital     float64            `json:"total_capital"`
    TotalDeployable  float64            `json:"total_deployable"`
    ReserveCash      float64            `json:"reserve_cash"`
    PositionSizes    map[string]float64 `json:"position_sizes"`
    Phase            domain.CapitalPhase `json:"phase"`
}

// CapitalAllocator distributes capital across positions based on phase limits
type CapitalAllocator struct {
    totalCapital float64
}

func NewCapitalAllocator(totalCapital float64) *CapitalAllocator {
    return &CapitalAllocator{totalCapital: totalCapital}
}

func (a *CapitalAllocator) Allocate(
    phaseConfig domain.CapitalPhaseConfig,
    recommendations []domain.Recommendation,
) CapitalAllocation {
    // Calculate deployable capital based on phase limit
    limit := phaseConfig.CapitalLimits[string(phaseConfig.CurrentPhase)]
    if limit == 0 {
        limit = 1.0
    }
    
    deployable := a.totalCapital * limit
    reserve := a.totalCapital - deployable
    
    // Equal-weight allocation across recommendations
    positionSizes := make(map[string]float64)
    if len(recommendations) > 0 {
        perPosition := deployable / float64(len(recommendations))
        for _, rec := range recommendations {
            positionSizes[rec.Symbol] = perPosition
        }
    }
    
    return CapitalAllocation{
        TotalCapital:    a.totalCapital,
        TotalDeployable: deployable,
        ReserveCash:     reserve,
        PositionSizes:   positionSizes,
        Phase:           phaseConfig.CurrentPhase,
    }
}

func (a *CapitalAllocator) ReallocateWithTax(
    phaseConfig domain.CapitalPhaseConfig,
    recommendations []domain.Recommendation,
    taxSnapshots []domain.TaxSnapshot,
) CapitalAllocation {
    base := a.Allocate(phaseConfig, recommendations)
    
    // Reduce deployable capital by expected tax
    var totalTax float64
    for _, ts := range taxSnapshots {
        totalTax += ts.TotalTax
    }
    
    base.TotalDeployable -= totalTax
    if base.TotalDeployable < 0 {
        base.TotalDeployable = 0
    }
    
    // Recalculate position sizes
    if len(recommendations) > 0 {
        perPosition := base.TotalDeployable / float64(len(recommendations))
        for _, rec := range recommendations {
            base.PositionSizes[rec.Symbol] = perPosition
        }
    }
    
    return base
}
