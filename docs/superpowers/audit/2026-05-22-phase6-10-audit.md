## 1. 執行摘要

### 系統現狀
Atlas 是一套**模擬優先、稽核導向**的台股投資研究系統，擁有 26,843 symbols、59,022 relationships、300 execution flows。系統的**風險計算能力已相當豐富**（VaR/CVaR/MaxDD/Component VaR/Kelly/ATR/HHI），但**風險執行層面仍偏弱**——多數風控停留在「計算與展示」，尚未形成完整的「偵測→決策→執行→驗證」閉環。

### 核心發現
| 領域 | 健康度 | 最關鍵問題 |
|------|--------|-----------|
| Phase 6 模擬交易 | 🟡 5/10 | narrative/industry context 未實際注入 pipeline |
| Phase 7 最新回測 | 🟡 5/10 | goroutine 洩漏風險、signal engine 分位數脆弱 |
| Phase 8 控制與稽核 | 🔴 3/10 | approve/reject 不進執行路徑、缺少審批鏈 |
| Phase 9 自我風控 | 🔴 3/10 | 計算豐富但執行薄弱、無統一 Risk Gate |
| Phase 10 跨板塊整合 | 🟡 4/10 | 多處違反統一架構、低軌衛星同步缺口 |

### 最優先修復
1. **統一 Risk Gate**（Phase 9）：將分散的風控訊號收斂為單一可執行引擎
2. **控制層閉環**（Phase 8）：讓 approve/reject 真正影響推薦流
3. **pipeline context 注入**（Phase 6）：修復 narrative/industry context 的 wiring
4. **回測可取消性**（Phase 7）：修復 goroutine 洩漏
5. **低軌衛星同步**（Phase 10）：補齊 supply_chain_graph.json 與前端字典

#### 6.3.4 API 合約

```go
// RiskDecision 風控決策
type RiskDecision struct {
    Phase      RiskPhase  `json:"phase"`       // pre_trade / in_trade / post_trade
    Verdict    Verdict    `json:"verdict"`      // ALLOW / REDUCE / BLOCK / HALT / ALERT_ONLY
    Action     RiskAction `json:"action"`
    Reason     string     `json:"reason"`
    Details    []RuleResult `json:"details"`    // 每條規則的判斷結果
    Timestamp  time.Time  `json:"timestamp"`
    Overridable bool      `json:"overridable"`  // 是否允許人工覆寫
}

type RiskPhase string
const (
    PhasePreTrade  RiskPhase = "pre_trade"
    PhaseInTrade   RiskPhase = "in_trade"
    PhasePostTrade RiskPhase = "post_trade"
)

type Verdict string
const (
    VerdictAllow     Verdict = "ALLOW"
    VerdictReduce    Verdict = "REDUCE"
    VerdictBlock     Verdict = "BLOCK"
    VerdictHalt      Verdict = "HALT"
    VerdictAlertOnly Verdict = "ALERT_ONLY"
)

type RiskAction struct {
    Type        ActionType `json:"type"`         // SELL / REDUCE / FREEZE / LIQUIDATE / NOTIFY
    TargetPct   float64    `json:"target_pct"`    // 目標倉位百分比
    Symbols     []string   `json:"symbols,omitempty"`
    Sectors     []string   `json:"sectors,omitempty"`
    Description string     `json:"description"`
}

type RuleResult struct {
    RuleName    string  `json:"rule_name"`
    Passed      bool    `json:"passed"`
    CurrentValue float64 `json:"current_value"`
    Threshold   float64 `json:"threshold"`
    Severity    string  `json:"severity"` // CRITICAL / WARNING / INFO
}

// RiskGate 統一風控入口介面
type RiskGate interface {
    // PreTradeCheck 下單前檢查
    PreTradeCheck(ctx context.Context, order OrderIntent, portfolio PortfolioState) (*RiskDecision, error)
    
    // InTradeCheck 持倉中檢查（每 N 秒輪詢或事件驅動）
    InTradeCheck(ctx context.Context, portfolio PortfolioState, marketData MarketSnapshot) (*RiskDecision, error)
    
    // PostTradeEval 盤後評估
    PostTradeEval(ctx context.Context, session SessionSummary) (*RiskDecision, error)
    
    // GetStatus 取得目前風控狀態
    GetStatus() RiskGateStatus
    
    // RegisterRule 註冊自訂規則
    RegisterRule(phase RiskPhase, rule RiskRule) error
}
```

#### 6.3.5 Pre-trade Gate 規則引擎

```go
// PreTradeGate 實作
type PreTradeGate struct {
    rules       []RiskRule
    riskManager *RiskManager
    varCalc     *VaRCalculator
    optimizer   *PortfolioOptimizer
}

func (g *PreTradeGate) Check(ctx context.Context, order OrderIntent, pf PortfolioState) (*RiskDecision, error) {
    decision := &RiskDecision{Phase: PhasePreTrade, Verdict: VerdictAllow}
    
    // 規則依優先序執行，第一個 BLOCK/HALT 即停止
    rules := []struct{
        name string
        fn   func() (bool, RuleResult)
    }{
        {"max_position_pct", g.checkMaxPosition},
        {"max_sector_exposure", g.checkSectorExposure},
        {"var_limit", g.checkVaRLimit},
        {"correlation_check", g.checkCorrelation},
        {"cash_buffer", g.checkCashBuffer},
        {"sector_ban", g.checkSectorBan},
        {"liquidity_check", g.checkLiquidity},
    }
    
    for _, rule := range rules {
        passed, result := rule.fn()
        decision.Details = append(decision.Details, result)
        
        if !passed && result.Severity == "CRITICAL" {
            decision.Verdict = VerdictBlock
            decision.Reason = result.RuleName
            break
        }
        if !passed {
            decision.Verdict = VerdictReduce
        }
    }
    
    return decision, nil
}

func (g *PreTradeGate) checkMaxPosition() (bool, RuleResult) {
    maxPct := config.GetParametersConfig().Optimizer.MaxPositionPct
    // 計算新倉位佔組合比例
    proposedPct := order.Notional / g.portfolio.TotalValue
    return proposedPct <= maxPct, RuleResult{
        RuleName: "max_position_pct",
        Passed:   proposedPct <= maxPct,
        CurrentValue: proposedPct,
        Threshold: maxPct,
        Severity: "CRITICAL",
    }
}
```

#### 6.3.6 In-trade Gate（即時監控）

```go
type InTradeGate struct {
    positions    map[string]*Position
    stopLosses   map[string]float64  // symbol → stop price
    takeProfits  map[string]float64  // symbol → target price
    trailingStops map[string]*TrailingStop
    volThreshold float64
}

// 監控循環（由 BackgroundTaskManager 排程，每 30 秒）
func (g *InTradeGate) Monitor(ctx context.Context) {
    ticker := time.NewTicker(30 * time.Second)
    defer ticker.Stop()
    
    for {
        select {
        case <-ctx.Done():
            return
        case <-ticker.C:
            for symbol, pos := range g.positions {
                price := g.marketData.GetPrice(symbol)
                
                // 1. 固定止損
                if stopPrice, ok := g.stopLosses[symbol]; ok && price <= stopPrice {
                    g.emitDecision(RiskDecision{
                        Phase: PhaseInTrade,
                        Verdict: VerdictHalt,
                        Action: RiskAction{Type: ActionSell, Symbols: []string{symbol}, TargetPct: 0},
                        Reason: fmt.Sprintf("stop-loss triggered: %.2f <= %.2f", price, stopPrice),
                    })
                }
                
                // 2. Trailing stop
                if ts, ok := g.trailingStops[symbol]; ok {
                    ts.UpdateHigh(price) // ATR-based trailing stop
                    if price <= ts.CurrentStop() {
                        g.emitDecision(RiskDecision{
                            Verdict: VerdictHalt,
                            Action: RiskAction{Type: ActionSell, Symbols: []string{symbol}},
                            Reason: "trailing stop triggered",
                        })
                    }
                }
                
                // 3. Volatility spike
                if g.volMonitor.IsSpike(symbol) {
                    g.emitDecision(RiskDecision{
                        Verdict: VerdictReduce,
                        Action: RiskAction{Type: ActionReduce, TargetPct: 0.5},
                        Reason: "volatility spike detected",
                    })
                }
            }
        }
    }
}
```

#### 6.3.7 Post-trade Gate（盤後評估）

```go
type PostTradeGate struct {
    sharpeMin     float64  // 低於此值 → CAUTIOUS
    drawdownMax   float64  // 超過此值 → DEFENSIVE
    drawdownHalt  float64  // 超過此值 → SUSPENDED
    consecutiveLoss int    // 連續虧損天數閾值
}

func (g *PostTradeGate) Evaluate(session SessionSummary) (*RiskDecision, error) {
    decision := &RiskDecision{Phase: PhasePostTrade, Verdict: VerdictAllow}
    
    // 1. Drawdown check
    if session.MaxDrawdown > g.drawdownHalt {
        decision.Verdict = VerdictHalt
        decision.Action = RiskAction{
            Type: ActionFreeze,
            Description: "Max drawdown exceeded halt threshold",
        }
        return decision, nil
    }
    
    if session.MaxDrawdown > g.drawdownMax {
        decision.Verdict = VerdictReduce
        decision.Action = RiskAction{
            Type: ActionReduce,
            TargetPct: 0.5,
            Description: "Reduce positions due to elevated drawdown",
        }
    }
    
    // 2. Sharpe degradation
    rollingSharpe := g.calcRollingSharpe(session, 20)
    if rollingSharpe < g.sharpeMin {
        decision.Verdict = VerdictReduce
        decision.Action.Type = ActionReduce
        decision.Action.TargetPct = max(decision.Action.TargetPct, 0.7)
    }
    
    // 3. Agent health → strategy adjustment
    for _, agent := range session.AgentHealth {
        if agent.Muted {
            decision.Details = append(decision.Details, RuleResult{
                RuleName: fmt.Sprintf("agent_%s_muted", agent.ID),
                Passed: false,
                Severity: "WARNING",
            })
        }
    }
    
    // 4. Regime shift → factor weight adjustment
    if g.regimeDetector.HasShifted() {
        newRegime := g.regimeDetector.CurrentRegime()
        decision.Action = RiskAction{
            Type: ActionNotify,
            Description: fmt.Sprintf("Regime shifted to %s. Consider adjusting factor weights.", newRegime),
        }
    }
    
    return decision, nil
}
```

// internal/orchestrator/system.go 修改

func (s *System) executeOrder(ctx context.Context, order OrderIntent) error {
    // === PRE-TRADE GATE ===
    decision, err := s.riskGate.PreTradeCheck(ctx, order, s.portfolio.State())
    if err != nil {
        return fmt.Errorf("risk gate error: %w", err)
    }
    
    if decision.Verdict == VerdictBlock || decision.Verdict == VerdictHalt {
        s.logger.Warn("order blocked by risk gate", "reason", decision.Reason)
        s.auditLog.Record(decision)
        return fmt.Errorf("order blocked: %s", decision.Reason)
    }
    
    if decision.Verdict == VerdictReduce {
        order.Notional *= decision.Action.TargetPct
    }
    
    // 執行下單...
    err = s.broker.PlaceOrder(ctx, order)
    
    // === 註冊 In-trade 監控 ===
    if err == nil {
        s.riskGate.RegisterPosition(order.Symbol, order.Price)
    }
    
    return err
}
